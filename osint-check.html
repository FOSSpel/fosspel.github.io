async function scan() { const val = document.getElementById('target').value.trim(); if(!val) return; const workerUrl = "https://fosspel-api.okuptcov4.workers.dev"; const resultsDiv = document.getElementById('geo-results'); const resultsBlock = document.getElementById('results'); resultsDiv.innerHTML = "Сканирование..."; resultsBlock.style.display = 'block'; try { const res = await fetch(`${workerUrl}/?target=${val}`); const data = await res.json(); if (data.error) throw new Error(data.error); let html = ''; // Если проверяли Email if (data.isEmail) { const badgeClass = data.disposable ? 'status-bad' : 'status-ok'; const badgeText = data.disposable ? 'ОПАСНО: ОДНОРАЗОВАЯ' : 'ЧИСТАЯ ПОЧТА'; html += `
Анализ Email: ${badgeText}
`; } // Данные об инфраструктуре html += `
IP: ${data.ip}
Страна: ${data.country}
Город: ${data.city}
Провайдер: ${data.isp}
ПОЛНЫЙ WHOIS ОТЧЕТ ↗
`; resultsDiv.innerHTML = html; } catch (err) { resultsDiv.innerHTML = `Ошибка: ${err.message}`; } }
