async function scan() { const val = document.getElementById('target').value.trim(); if(!val) return; const btn = document.getElementById('scanBtn'); const resBlock = document.getElementById('results'); const workerUrl = "https://fosspel-api.okuptcov4.workers.dev"; btn.innerText = "Processing..."; resBlock.style.display = 'none'; try { const res = await fetch(`${workerUrl}/?target=${encodeURIComponent(val)}&t=${Date.now()}`); const data = await res.json(); // Почта let emailHtml = ''; if (data.isEmail) { const statusClass = data.disposable ? 'status-bad' : 'status-ok'; const statusText = data.disposable ? 'ОДНОРАЗОВАЯ' : 'ЧИСТАЯ'; emailHtml = `
Анализ почты: ${statusText}
`; } document.getElementById('emailInfo').innerHTML = emailHtml; // Гео данные document.getElementById('geoInfo').innerHTML = `
IP Адрес: ${data.ip}
Страна: ${data.country}
Город: ${data.city}
Провайдер: ${data.isp}
`; // WHOIS document.getElementById('whoisLink').href = `https://whois.domaintools.com/${data.domain}`; document.getElementById('whoisLink').innerText = `WHOIS ОТЧЕТ ДЛЯ ${data.domain.toUpperCase()} ↗`; resBlock.style.display = 'block'; } catch (err) { alert("Ошибка: " + err.message); } finally { btn.innerText = "Scan"; } }
