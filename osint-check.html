async function scan() { const inputField = document.getElementById('target'); if (!inputField) return; let val = inputField.value.trim(); if (!val) return; const btn = document.getElementById('scanBtn'); const resBlock = document.getElementById('results'); // ✅ Убраны лишние пробелы в URL const workerUrl = "https://fosspel-api.okuptcov4.workers.dev"; // Сохраняем исходный текст кнопки для восстановления const originalBtnText = btn.innerText; btn.innerText = "wait..."; if (resBlock) resBlock.style.display = 'none'; try { // Добавляем timestamp, чтобы избежать кэширования const res = await fetch(`${workerUrl}/?target=${encodeURIComponent(val)}&t=${Date.now()}`); if (!res.ok) { throw new Error(`HTTP error! status: ${res.status}`); } const data = await res.json(); // ✅ Безопасная работа с почтой const emailInfo = document.getElementById('emailInfo'); if (emailInfo && data.isEmail) { const statusClass = data.disposable ? 'status-bad' : 'status-ok'; const statusText = data.disposable ? 'ОДНОРАЗОВАЯ' : 'ЧИСТАЯ'; emailInfo.innerHTML = `
Почта: ${statusText}
`; } else if (emailInfo) { emailInfo.innerHTML = ''; // Очищаем, если не email } // ✅ Безопасная работа с гео-данными const geoInfo = document.getElementById('geoInfo'); if (geoInfo) { geoInfo.innerHTML = `
IP Адрес: ${data.ip ?? "N/A"}
Страна: ${data.country ?? "N/A"}
Город: ${data.city ?? "N/A"}
Провайдер: ${data.isp ?? "N/A"}
`; } // ✅ Безопасная работа с WHOIS-ссылкой (без пробелов!) const whoisLink = document.getElementById('whoisLink'); if (whoisLink && data.domain) { whoisLink.href = `https://whois.domaintools.com/${encodeURIComponent(data.domain)}`; whoisLink.innerText = `ПОЛНЫЙ WHOIS ОТЧЕТ ДЛЯ ${data.domain.toUpperCase()} ↗`; } if (resBlock) resBlock.style.display = 'block'; } catch (err) { console.error("Scan error:", err); // ← Смотрите в консоль (F12) alert("Ошибка: " + err.message); } finally { if (btn) btn.innerText = originalBtnText || "Scan"; } }
