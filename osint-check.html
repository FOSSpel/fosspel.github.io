async function scan() { const inputField = document.getElementById('target'); let val = inputField.value.trim(); if(!val) return; const btn = document.getElementById('scanBtn'); const resBlock = document.getElementById('results'); const workerUrl = "https://fosspel-api.okuptcov4.workers.dev"; btn.innerText = "wait..."; resBlock.style.display = 'none'; try { const res = await fetch(`${workerUrl}/?target=${encodeURIComponent(val)}`); const data = await res.json(); // Если API не нашло домен (success: false), выводим ошибку if (!data.success) { throw new Error("Домен не найден или защищен."); } // Вывод почты let emailHtml = ''; if (data.isEmail) { const statusClass = data.disposable ? 'status-bad' : 'status-ok'; const statusText = data.disposable ? 'ОДНОРАЗОВАЯ' : 'ЧИСТАЯ'; emailHtml = `
Почта: ${statusText}
`; } document.getElementById('emailInfo').innerHTML = emailHtml; // Вывод Гео document.getElementById('geoInfo').innerHTML = `
IP Адрес: ${data.ip}
Страна: ${data.country}
Город: ${data.city}
Провайдер: ${data.isp}
`; // Прямая ссылка на WHOIS для nare37.com document.getElementById('whoisLink').href = `https://whois.domaintools.com/${data.domain}`; document.getElementById('whoisLink').innerText = `ПОЛНЫЙ WHOIS ОТЧЕТ ДЛЯ ${data.domain.toUpperCase()} ↗`; resBlock.style.display = 'block'; } catch (err) { alert("Ошибка: " + err.message); } finally { btn.innerText = "Scan"; } }
