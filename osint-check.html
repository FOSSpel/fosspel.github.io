async function scan() { const val = document.getElementById('target').value.trim(); if(!val) return; const workerUrl = "https://fosspel-api.okuptcov4.workers.dev/"; try { const res = await fetch(`${workerUrl}/?target=${val}`); const data = await res.json(); if(data.error) throw new Error(data.error); // Теперь данные ТОЧНО придут, так как Worker обходит CORS document.getElementById('geo-results').innerHTML = `
IP:${data.geo.ip || data.dns}
Страна:${data.geo.country_name || 'N/A'}
ISP:${data.geo.org || 'N/A'}
`; } catch (err) { alert("Ошибка: Проверьте URL воркера или лимиты."); } }
