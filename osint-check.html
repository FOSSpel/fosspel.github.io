async function scan() { const val = document.getElementById('target').value.trim(); if(!val) return; // Скрываем кнопку, показываем лоадер (если есть) const workerUrl = "https://fosspel-api.okuptcov4.workers.dev"; try { const res = await fetch(`${workerUrl}/?target=${val}`); const data = await res.json(); if (data.error) { alert("Ошибка воркера: " + data.error); return; } // Вывод данных document.getElementById('geo-results').innerHTML = `
IP: ${data.ip}
Страна: ${data.country}
Город: ${data.city}
Провайдер: ${data.org}
`; // Показываем блок результатов document.getElementById('results').style.display = 'block'; } catch (err) { console.error(err); alert("Не удалось связаться с воркером. Проверьте консоль (F12)."); } }
